# How a comparison works

The interactive flowchart below follows a single `get_irmsd` call through the Fortran backend:
the canonical atom ranks, the alignment and LSAP assignment in `min_rmsd`, the bond-preserving selection among candidate assignments, and the centre walk that builds one of those candidates.
Its last section animates three worked examples on small 2D toy molecules.
Boxes with a link jump to the section that explains them.

<script>
(function () {
  // size the frame to its content and pass the docs light/dark mode on to it
  var frame = document.getElementById("irmsd-flowchart");
  frame.addEventListener("load", function () {
    var doc;
    try { doc = frame.contentDocument; } catch (e) { return; }
    if (!doc) return;
    var fit = function () { frame.style.height = doc.documentElement.scrollHeight + "px"; };
    var theme = function () {
      var cls = document.documentElement.classList;
      if (cls.contains("dark")) doc.documentElement.setAttribute("data-theme", "dark");
      else if (cls.contains("light")) doc.documentElement.setAttribute("data-theme", "light");
      else doc.documentElement.removeAttribute("data-theme");
    };
    theme();
    new MutationObserver(theme).observe(document.documentElement, { attributes: true, attributeFilter: ["class"] });
    if (window.ResizeObserver) new ResizeObserver(fit).observe(doc.body);
    fit();
  });
})();
</script>

The page is generated by `docs/flowchart/build.py` from `template.html` and the example frames in `examples.json`, which `examples.py` computes with a Python transcription of `min_rmsd` and the centre walk.
The documentation build runs `build.py`; after changes to the examples, run `python docs/flowchart/examples.py` first.
